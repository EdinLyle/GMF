// 主进程：本地模型管理器
class LocalModelManager {
  private models: Map<string, LocalModelConnection> = new Map();
  
  async discoverLocalModels(): Promise<ModelInfo[]> {
    // 自动发现本地 Ollama 模型
    const ollamaModels = await this.scanOllama();
    
    // 自动发现本地 llama.cpp 服务
    const llamaCppModels = await this.scanLlamaCpp();
    
    // 用户手动添加的 API 端点
    const customModels = await this.loadCustomEndpoints();
    
    return [...ollamaModels, ...llamaCppModels, ...customModels];
  }
  
  async runComparisonTest(
    prompt: string,
    selectedModels: string[]
  ): Promise<ComparisonResult> {
    // 并行测试（Electron 主进程异步能力）
    const results = await Promise.all(
      selectedModels.map(async modelId => {
        const model = this.models.get(modelId);
        const startTime = performance.now();
        
        try {
          const response = await model.generate(prompt);
          return {
            modelId,
            success: this.evaluateSuccess(response),
            latency: performance.now() - startTime,
            response: response.substring(0, 500),
            tokenCount: response.length / 4  // 估算
          };
        } catch (error) {
          return { modelId, error: error.message, success: false };
        }
      })
    );
    
    return this.aggregateComparison(results);
  }
}

### 3. **离线语义分析引擎（基于 Transformers.js）**

**为什么可行**：使用 ONNX 模型，完全本地推理，无需网络

```typescript
// 主进程：使用 Transformers.js（纯 Node.js）
import { pipeline } from '@xenova/transformers';

class LocalSemanticAnalyzer {
  private embedder: any;
  private classifier: any;
  
  async initialize() {
    // 下载一次后永久本地缓存（~80MB）
    this.embedder = await pipeline(
      'feature-extraction',
      'Xenova/all-MiniLM-L6-v2',
      { quantized: true }  // 量化模型，更小更快
    );
    
    // 轻量级分类器（本地微调或零样本）
    this.classifier = await pipeline(
      'zero-shot-classification',
      'Xenova/mobilebert-uncased-mnli'
    );
  }
  
  async analyze(prompt: string): Promise<SemanticAnalysis> {
    // 1. 生成 Embedding（本地）
    const embedding = await this.embedder(prompt);
    
    // 2. 与本地攻击向量库对比（SQLite 存储）
    const similarAttacks = await this.findSimilarAttacks(embedding);
    
    // 3. 零样本分类（本地）
    const classification = await this.classifier(prompt, [
      'prompt injection',
      'jailbreak attempt', 
      'normal query',
      'data exfiltration'
    ]);
    
    return {
      embedding: embedding.data,
      similarityScore: Math.max(...similarAttacks.map(a => a.score)),
      classification: classification.labels[0],
      confidence: classification.scores[0],
      similarCases: similarAttacks.slice(0, 5)
    };
  }
  
  // 本地攻击向量库（SQLite）
  private async findSimilarAttacks(embedding: Tensor): Promise<SimilarAttack[]> {
    // 使用本地 SQLite + 向量相似度搜索
    const db = await sqlite.open('/gmf_data/attack_vectors.db');
    
    // 预先计算的 attack embeddings
    const candidates = await db.all(`
      SELECT id, prompt, embedding 
      FROM attack_vectors 
      WHERE category IN ('jailbreak', 'injection')
    `);
    
    // 本地计算余弦相似度
    return candidates
      .map(c => ({
        ...c,
        score: cosineSimilarity(embedding, JSON.parse(c.embedding))
      }))
      .sort((a, b) => b.score - a.score);
  }
}
```

// 主进程：本地合规检查引擎
class LocalComplianceChecker {
  private frameworks: Map<string, ComplianceFramework>;
  
  constructor() {
    // 内置规则库（JSON 文件，可更新）
    this.frameworks = new Map([
      ['OWASP_LLM_TOP_10_2025', owaspRules],
      ['NIST_AI_RMF_1.0', nistRules],
      ['GB_T_XXXX', chineseStandardRules]  // 可添加国标
    ]);
  }
  
  async check(prompts: Prompt[], frameworkId: string): Promise<ComplianceReport> {
    const framework = this.frameworks.get(frameworkId);
    const results = [];
    
    for (const prompt of prompts) {
      for (const rule of framework.rules) {
        // 本地规则检查（正则、关键词、简单语义）
        const passed = this.evaluateRule(prompt, rule);
        
        results.push({
          promptId: prompt.id,
          ruleId: rule.id,
          ruleName: rule.name,
          passed,
          evidence: passed ? null : this.getFailureEvidence(prompt, rule),
          severity: rule.severity
        });
      }
    }
    
    return this.generateReport(results, framework);
  }
  
  // 本地生成 PDF（使用 Puppeteer 或纯 HTML）
  async exportPDF(report: ComplianceReport, template: string): Promise<Buffer> {
    // Electron 内置 Chromium，可直接用
    const win = new BrowserWindow({ show: false });
    await win.loadURL(`data:text/html,${this.renderHTML(report, template)}`);
    
    const pdf = await win.webContents.printToPDF({
      marginsType: 1,
      printBackground: true,
      pageSize: 'A4'
    });
    
    win.close();
    return pdf;
  }
}
```

// 主进程：本地知识库
class LocalKnowledgeBase {
  private db: Database;
  
  async initialize() {
    this.db = await open({
      filename: '/gmf_data/knowledge.db',
      driver: sqlite3.Database
    });
    
    // 启用 FTS5 全文搜索
    await this.db.exec(`
      CREATE VIRTUAL TABLE IF NOT EXISTS cases USING fts5(
        title,
        content,
        tags,
        category,
        timestamp
      );
    `);
  }
  
  // 添加案例（用户研究或导入）
  async addCase(caseData: CaseData): Promise<string> {
    const id = uuidv4();
    
    await this.db.run(`
      INSERT INTO cases (rowid, title, content, tags, category, timestamp)
      VALUES (?, ?, ?, ?, ?, ?)
    `, [id, caseData.title, caseData.content, 
        caseData.tags.join(','), caseData.category, Date.now()]);
    
    return id;
  }
  
  // 全文搜索
  async search(query: string, filters: SearchFilters): Promise<Case[]> {
    // 支持中文分词（使用 simple 分词器或自定义）
    const sql = `
      SELECT rowid, title, snippet(cases, 1, '<mark>', '</mark>', '...', 32) as snippet
      FROM cases
      WHERE cases MATCH ?
      ${filters.category ? 'AND category = ?' : ''}
      ORDER BY rank
      LIMIT 50;
    `;
    
    return await this.db.all(sql, [query, filters.category].filter(Boolean));
  }
  
  // 相似案例推荐（基于本地 embedding）
  async findSimilar(caseId: string): Promise<Case[]> {
    const target = await this.getEmbedding(caseId);
    
    const allCases = await this.db.all('SELECT rowid, embedding FROM cases');
    
    return allCases
      .map(c => ({
        ...c,
        similarity: cosineSimilarity(target, JSON.parse(c.embedding))
      }))
      .sort((a, b) => b.similarity - a.similarity)
      .slice(1, 6);  // 排除自己，取前 5
  }
}
```
